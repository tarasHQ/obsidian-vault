<%*
// ================================================================
// TEMPLATE: Reflection Capture
//
// LLM RULE:
// DON'T USE <> TAGS in comments. It breaks further rendering of markdown in obsidian. 
//
// WHAT THIS DOES:
// Routes a reflection note to either:
//   - Project Root _reflections/
//   - Phase-specific phases/NNN/_reflections/
//   - Canvas-based project root (ThePlan) _reflections/
//
// TWO MODES:
// - MANUAL: Uses the interactive workflow below.
// - AUTO: Called by [T]phase.md for a note already at its final path; skips
//   prompts, moving, and title editing while rendering the same content.
//
// HOW IT WORKS:
// 1. Gathers all folders ending with the configured reflection suffix.
// 2. Formats readable labels for the selector dropdown.
// 3. Prompts user to choose destination folder.
// 4. Resolves collisions (Untitled Reflection, Untitled Reflection 2, etc.).
// 5. Moves note and triggers Obsidian's title editor for immediate renaming.
// ================================================================

// ================================================================
// 0. CONFIG — the single place to adjust names / paths / prompts.
//    NOTE: Other templates have their own CFG block; if a SHARED value changes
//    (phasesDirName, reflectionFolderSuffix, ...), update it there as well.
//    autoCaptureContext must match [T]phase.md and [T]ideas.md.
// ================================================================
const CFG = {
    // --- Folder discovery & structure
    reflectionFolderSuffix: "_reflections", // folders with this suffix are offered as destinations
    phasesDirName: "phases", // phase folders live at project root/phases/NNN

    // --- Suggester labels & prompts
    rootLabel: "[Project Root]", // label prefix for project-root destination
    phaseLabelPrefix: "[Phase ", // label prefix for phase destinations: "[Phase 002] path"
    promptSelectDestination: "Select reflection destination:",

    // --- Default note naming (collision-safe: "... 2", "... 3", ...)
    defaultNoteName: "Untitled Reflection",
    counterSeparator: " ",

    // --- Programmatic generation
    autoCaptureContext: "__phaseCaptureAuto", // applies only to the specified child file

    // --- UX / environment
    titleEditCommand: "workspace:edit-file-title", // command that focuses the inline title editor
    titleEditDelayMs: 200, // wait after move before focusing the title
    ctimeFormat: "YYYY-MM-DD HH:mm",
};

const escapeRegExp = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
// Matches the phase number segment inside a path: .../phasesDirName/NNN/...
const phaseSegmentRegex = new RegExp(`${escapeRegExp(CFG.phasesDirName)}/([^/]+)`);

// Exact-path matching prevents automatic mode from affecting an unrelated note.
const autoContext = window[CFG.autoCaptureContext];
const isAutoGenerated = autoContext?.targetPath === tp.config.target_file.path;
let targetFolder;
let finalName;

if (isAutoGenerated) {
    // create_new has already placed this note at its final destination.
    targetFolder = tp.config.target_file.parent.path;
} else {
    // 1. Gather all folders ending with the configured reflection suffix
    const reflectionFolders = app.vault.getAllLoadedFiles()
        .filter(item => item.children && item.name.endsWith(CFG.reflectionFolderSuffix))
        .sort((a, b) => {
            // Keep project-root reflections first, then sort phase folders naturally
            const aIsRoot = !a.path.includes(CFG.phasesDirName);
            const bIsRoot = !b.path.includes(CFG.phasesDirName);
            if (aIsRoot && !bIsRoot) return -1;
            if (!aIsRoot && bIsRoot) return 1;
            return a.path.localeCompare(b.path, undefined, { numeric: true, sensitivity: "base" });
        });

    if (reflectionFolders.length === 0) {
        new Notice(`No '${CFG.reflectionFolderSuffix}' folders found in vault!`);
        return;
    }

    // 2. Format readable labels for suggester
    const folderLabels = reflectionFolders.map(f => {
        const phaseMatch = f.path.match(phaseSegmentRegex);
        if (phaseMatch) {
            return `${CFG.phaseLabelPrefix}${phaseMatch[1]}] ${f.path}`;
        }
        return `${CFG.rootLabel} ${f.path}`;
    });

    // 3. User selects destination folder
    const selectedFolder = await tp.system.suggester(
        folderLabels,
        reflectionFolders,
        false,
        CFG.promptSelectDestination
    );

    if (!selectedFolder) return;

    // 4. Resolve unique placeholder filename if one already exists
    targetFolder = selectedFolder.path.replace(/\/$/, "");
    let baseName = CFG.defaultNoteName;
    finalName = baseName;
    let counter = 1;

    while (app.vault.getAbstractFileByPath(`${targetFolder}/${finalName}.md`)) {
        counter++;
        finalName = `${baseName}${CFG.counterSeparator}${counter}`;
    }
}

if (!isAutoGenerated) {
    // 5. Move note to the selected folder
    await tp.file.move(`${targetFolder}/${finalName}`);

    // 6. Focus the note title in Obsidian after leaf render
    setTimeout(() => {
        app.commands.executeCommandById(CFG.titleEditCommand);
    }, CFG.titleEditDelayMs);
}
-%>
